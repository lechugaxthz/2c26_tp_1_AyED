% CORRECCIÓN: "Los enum no son strings, no van entre comillas. emisor != emision."
\aux{millasDisponibles}{h: \Tipo{historial}}{\R}{
    \sum_{k=0}^{|h|-1} \Tipo{IfThenElse}(h[k].transaccion = \comillas{emision} \lor \\
    \tab\tab h[k].transaccion = \comillas{canje}, \\
    \tab\tab - h[k].millas, \\
    \tab\tab h[k].millas)
}

\aux{millasAcumuladas}{h: \Tipo{historial}}{\R}{
    \sum_{k=0}^{|h|-1} \IfThenElse{h[k].transaccion = \comillas{emision} \lor h[k].transaccion = \comillas{canje}}{0}{h[k].millas}
}

% CORRECCIÓN: "Repetida" (es igual a totalMillasCanjeadas)
\aux{millasCanjeadas}{h: \Tipo{historial}}{\R}{
    \sum_{k=0}^{|h|-1} \IfThenElse{h[k].transaccion = \comillas{canje}}{h[k].millas}{0}
}

% CORRECCIÓN: "es un pred porque devuelve un bool."
% CORRECCIÓN: "Mal uso de cuantificadores. La sintaxis es ((cuantificador) nombreDelArgumento: TipoDeDato).
%              Lo están aplicando a una lista y no funciona así."
% CORRECCIÓN: "faltan paréntesis. Esto aparece más abajo. Tenían que borrarlo."
\aux{sonExclusivos}{e1: \Tipo{emision}, e2: \Tipo{emision}, u1: \Tipo{id}, u2: \Tipo{id}}{\Tipo{bool}}{
    ((|e1|, |e2| > 0) \land (\forall j: e1) j = u2) \land ((\forall k: e2) k = u1)
}

\aux{totalMillasPorVuelo}{h: \Tipo{historial}}{\R}{
    \sum_{i=0}^{|h|-1} \IfThenElse{h[i].transaccion = \comillas{vuelo}}{h[i].millas}{0}
}

% CORRECCIÓN: "Repetida" (es igual a millasCanjeadas)
\aux{totalMillasCanjeadas}{h: \Tipo{historial}}{\R}{
    \sum_{i=0}^{|h|-1} \IfThenElse{h[i].transaccion = \comillas{canje}}{h[i].millas}{0}
}

% Antes era totalMillasEmitidasRecibidas y sumaba emitidas y recibidas juntas (se perdía información).
% Ahora se separa en dos auxiliares: uno para las emitidas y otro para las recibidas.
\aux{totalMillasEmitidas}{h: \Tipo{historial}}{\R}{
    \sum_{i=0}^{|h|-1} \IfThenElse{h[i].transaccion = \Tipo{emisor}}{h[i].millas}{0}
}

\aux{totalMillasRecibidas}{h: \Tipo{historial}}{\R}{
    \sum_{i=0}^{|h|-1} \IfThenElse{h[i].transaccion = \Tipo{receptor}}{h[i].millas}{0}
}

\noindent\texttt{aux millasGanadasPorVuelo}$(distancia: \R, historial: \Tipo{historial}, factorPromo: \Tipo{factor}):$ \\
$\R = distancia \cdot factorCategoriaUsuario(historial) \cdot factorPromo$

\medskip

% CORRECCIÓN: "Rebuscado. Los auxiliares no tienen tipo pasaje de parámetro" (sobra el "in")
\aux{factorCategoriaUsuario}{\In h: \Tipo{historial}}{\R}{
    (1 + factorCondicional(millasAcumuladas(h), 0.5, 0, 10000) + \\
    factorCondicional(millasAcumuladas(h), 0.5, 0, 50000)) \\
    \cdot factorCondicional(millasAcumuladas(h), 0, 1, 1000000)
}

\aux{factorCondicional}{m: millas, factor1: \R, factor2: \R, rango: \N}{\R}{
    \IfThenElse{m \geq rango}{factor1}{factor2}
}

% CORRECCIÓN: "indexa usuarios[ranking[i]] sin que nada garantice que ranking[i] sea un usuario. Se indefine."
\pred{esRankingOrdenado}{ranking: \seq{\Tipo{id}}, usuarios: \dict{\Tipo{id}}{\langle \Tipo{historial}, \Tipo{emision} \rangle}}{
    \paraTodo{i}{\Z}{0 \leq i < |ranking| - 1 \implicaLuego \\
    millasDisponibles(usuarios[ranking[i]].historial) \\
    \geq millasDisponibles(usuarios[ranking[i+1]].historial)}
}

\pred{tieneALosTopN}{ranking: \seq{\Tipo{id}}, usuarios: \dict{\Tipo{id}}{\langle \Tipo{historial}, \Tipo{emision} \rangle}}{
    (\forall i, j: \Tipo{id})\ ((j \in ranking \land j \in \claves{usuarios} \land i \in \claves{usuarios} \land i \notin \\
    ranking) \implicaLuego \\
    millasDisponibles(usuarios[i].historial) \leq millasDisponibles(usuarios[j].historial))
}

\pred{esPromoValida}{promo: \Tipo{struct}\langle \Tipo{factor}, \Tipo{tope}, \Tipo{tieneTope} \rangle, millasAGanar: \Tipo{millas}}{
    (\lnot promo.tieneTope \implica promo.tope = 0) \land \\
    \tab (promo.tieneTope \implica millasAGanar \leq promo.tope)
}

\pred{sonExclusivos}{emisiones_1: \seq{\Tipo{id}}, emisiones_2: \seq{\Tipo{id}}, u_1: \Tipo{id}, u_2: \Tipo{id}}{
    esUnicoEmisor(emisiones_1, u_2) \land esUnicoEmisor(emisiones_2, u_1)
}

\pred{esUnicoEmisor}{emisiones: \seq{\Tipo{id}}, emisor: \Tipo{id}}{
    |emisiones| > 0 \land \paraTodo{i}{\Z}{0 \leq i < |emisiones| \implicaLuego emisiones[i] = emisor}
}
