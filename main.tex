\section{Especificación del TAD}

\begin{tad}{AirMiles}

\medskip

\obs{usuarios}{\dicc{idUsuario}{infoUsuario}}\\
\obs{promos}{\dicc{codigo}{\langle factorPromocional,tope \rangle}}

\medskip
\medskip

\begin{proc}{registrarUsuario}
{
\Inout a: \Tipo{AirMiles},
}{}
    \requiereCorto{a = A_0}
    %\yLuego u \notin \claves{a_0.usuarios}}
    \asegura{  \existe{k}{idUsuario}{k \notin \claves{A_0.usuarios}} \land \\
    a.usuarios = \setKey{A_0.usuarios}{k}{\literal{\vacio, \vacio}} \land \\
              a.promos = A_0.promos }
\end{proc}

% a,usuarios[u1].historial

\begin{proc}{transferirMillas}
{
\In u_1: idUsuario, \In u_2: idUsuario, \\ \tab
\In m: millas, \Inout a: AirMiles
}{}
    \requiere{ a = A_0 \yLuego u_1 \in \claves{A_0.usuarios} \land u_2 \in \claves{A_0.usuarios} \yLuego \\
               u_1 \neq u_2 \land m > 0 \land millasDisponibles(A_0[u_1].historial) \geq m \\
               \land factorCategoriaUsuario(u_2) \neq 0}
    \asegura{ \claves{a} = \claves{A_0} \land \\
              a.usuarios[u_1].historial = A_0.usuarios[u_1].historial \concat \literal{\literal{emisor, m, NoPromo}} \land \\
              a.usuarios[u_1].emision = A_0.usuarios[u_1].emision \concat \literal{u_2} \land \\
              a.usuarios[u_2].historial = A_0.usuarios[u_2].historial \concat \literal{\literal{receptor, m, NoPromo}} \land \\
              a.usuarios[u_2].emision = A_0.usuarios[u_2].emision \land \\
              %restrinjo que no se cambie nada mas del resto de los usuarios%
              \paraTodo{k}{idUsuario}{(k \in \claves{A_0.usuarios} \land k \neq u_1 \land k \neq u_2) \implicaLuego \\
              a.usuarios[k] = A_0.usuarios[k]} }
\end{proc}


\begin{proc}{resumenTransacciones}
{
\In u: idUsuario,
\In a: AirMiles 
}{
\tupla{\R, \R, \R}
}
    \requiereCorto{u \in \claves{a.usuarios}}
    \asegura{ res = \literal{ totalMillasPorVuelo(a.usuarios[u].historial), \\
              totalMillasCanjeadas(a.usuarios[u].historial), \\
              totalMillasEmitidasRecibidas(a.usuarios[u].historial) } }
\end{proc}


\begin{proc}{elQueTieneMasMillas}
{
\In a: \Tipo{AirMiles}}{idUsuario}
    \requiereCorto{\claves{a.usuarios} \neq \emptyset}
    \asegura{ res \in \claves{a.usuarios} \land \\
              \paraTodo{k}{usuario}{k \in \claves{a.usuarios} \implicaLuego \\
              millasAcumuladas(a.usuarios[k].historial) \leq \\ millasAcumuladas(a.usuarios[res].historial)} }
\end{proc}

\medskip

\auxCorto{millasAcumuladas}{h: historial}{\R}{\sum_{i=0}^{\longitud{h}-1} h[i].monto}

\aux{totalMillasPorVuelo}{h: historial}{\R}{
    \sum_{i=0}^{\longitud{h}-1} \IfThenElse{h[i].tipo = vuelo}{h[i].monto}{0}
}

\aux{totalMillasCanjeadas}{h: historial}{\R}{
    \sum_{i=0}^{\longitud{h}-1} \IfThenElse{h[i].tipo = canje}{h[i].monto}{0}
}

% Cuenta tanto las millas emitidas como las recibidas.
\aux{totalMillasEmitidasRecibidas}{h: historial}{\R}{
    \sum_{i=0}^{\longitud{h}-1} \IfThenElse{h[i].tipo = emisor \lor h[i].tipo = receptor}{h[i].monto}{0}
}


\end{tad}
